cmake_minimum_required(VERSION 3.24)
project(ZynumInstalledConsumer LANGUAGES C CXX)
include(CTest)
set(ZYNUM_TEST_COMPONENTS "" CACHE STRING "Explicit package components")
if(ZYNUM_TEST_OPTIONAL_STANDALONE)
    find_package(Zynum CONFIG REQUIRED COMPONENTS core OPTIONAL_COMPONENTS standalone)
elseif(ZYNUM_TEST_COMPONENTS)
    find_package(Zynum CONFIG REQUIRED COMPONENTS ${ZYNUM_TEST_COMPONENTS})
else()
    find_package(Zynum CONFIG REQUIRED)
endif()
if(ZYNUM_TEST_OPTIONAL_STANDALONE AND Zynum_standalone_FOUND)
    message(FATAL_ERROR "This optional-component fixture expects standalone to be unavailable")
endif()
if(NOT DEFINED Zynum_VERSION OR Zynum_VERSION STREQUAL "")
    message(FATAL_ERROR "Unexpected source package version: ${Zynum_VERSION}")
endif()
if(NOT "standalone" IN_LIST ZYNUM_TEST_COMPONENTS AND TARGET Zynum::standalone)
    message(FATAL_ERROR "Default/core-only package request exposed standalone")
endif()
foreach(language c cpp)
    add_executable(package_${language} consumer.${language})
    target_link_libraries(package_${language} PRIVATE Zynum::core)
    target_compile_features(package_${language} PRIVATE c_std_11 cxx_std_20)
    target_compile_options(package_${language} PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME package_${language} COMMAND package_${language})
endforeach()
# Reuse production API/provider lifetime and Linux ABI regression fixtures,
# compiled against only the installed target; no add_subdirectory of Zynum.
foreach(fixture native_context.c native_complex_abi.c native_complex_abi.cpp)
    string(REPLACE "." "_" name "${fixture}")
    add_executable(${name} "../${fixture}")
    target_link_libraries(${name} PRIVATE Zynum::core)
    target_compile_features(${name} PRIVATE c_std_11 cxx_std_20)
    target_compile_options(${name} PRIVATE -Wall -Wextra -Werror -UNDEBUG -fno-fast-math)
    add_test(NAME ${name} COMMAND ${name})
endforeach()
if("standalone" IN_LIST ZYNUM_TEST_COMPONENTS)
    add_executable(package_standalone ../standalone.c)
    target_link_libraries(package_standalone PRIVATE Zynum::standalone)
    target_compile_features(package_standalone PRIVATE c_std_11)
    target_compile_options(package_standalone PRIVATE -Wall -Wextra -Werror -UNDEBUG)
    add_test(NAME package_standalone COMMAND package_standalone)
endif()
